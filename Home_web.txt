@code {
    private string nuevoPrompt = "Clasificar intención de texto con LLM";
    private List<LogItem> historial = new();

    private async Task ProcesarPeticion()
    {
        if (string.IsNullOrWhiteSpace(nuevoPrompt)) return;

        long relojAlEnviar = IaClient.RelojLogicoCliente + 1; // Próximo tick de emisión
        var respuesta = await IaClient.EnviarInferenciaAsync(nuevoPrompt);

        if (respuesta != null)
        {
            historial.Insert(0, new LogItem(relojAlEnviar, nuevoPrompt, respuesta, IaClient.RelojLogicoCliente));
        }
    }

    private record LogItem(long RelojEnvio, string Prompt, InferenciaResponse Respuesta, long RelojPostRecepcion);
}
